    def _authority_governance_policy(self, prepared_class: str, certified_class: str, lifecycle_state: str) -> dict[str, Any]:
        promotion_blockers: list[str] = []
        demotion_risks: list[str] = []
        freeze_triggers = [
            "missing_evidence",
            "unknown_runtime_truth",
            "snapshot_source_mismatch",
            "approval_packet_missing_or_stale",
            "restore_barrier_missing_or_stale",
            "trust_or_prediction_unknown",
        ]
        if AUTHORITY_CLASS_RANK[prepared_class] > AUTHORITY_CLASS_RANK[certified_class]:
            promotion_blockers.append("prepared_authority_exceeds_certified_evidence")
        if lifecycle_state in {"FROZEN", "REVOKED"}:
            promotion_blockers.append(f"authority_{lifecycle_state.lower()}")
        if lifecycle_state in {"PREPARED", "DEMOTED", "FROZEN", "REVOKED"}:
            demotion_risks.append(f"lifecycle_state_{lifecycle_state.lower()}")
        return {
            "promotion": {
                "owner": "operator_governance_existing_policy",
                "executor": "tools/v7-users-autoswitch",
                "requires_explicit_operator_approval": True,
                "requires_runtime_truth_known": True,
                "requires_successful_outcomes": True,
                "requires_verification_evidence": True,
                "requires_rollback_capability_evidence": True,
                "requires_trust_prediction_recommendation_feedback": True,
                "eligible": not promotion_blockers,
                "blockers": promotion_blockers,
            },
            "demotion": {
                "owner": "operator_governance_existing_policy",
                "executor": "tools/v7-users-autoswitch",
                "triggers": [
                    "rollback_required_or_failed",
                    "verification_failure",
                    "prediction_degradation",
                    "trust_degradation",
                    "service_instability",
                    "operator_policy_reduction",
                ],
                "risks": demotion_risks,
            },
            "freeze": {
                "owner": "operator_governance_existing_policy",
                "executor": "tools/v7-users-autoswitch",
                "triggers": freeze_triggers,
                "action": "set_effective_runtime_budget_to_zero_until_truth_or_evidence_is_known",
            },
        }

    def _authority_budget_gate(self, selected: list[dict[str, Any]]) -> tuple[list[dict[str, Any]], dict[str, Any]]:
        policy = dict(self.authority_budget_policy)
        authority_class = normalize_authority_class(policy.get("authority_class"))
        prepared_class = normalize_authority_class(policy.get("prepared_authority_class") or authority_class)
        certified_class = normalize_authority_class(policy.get("certified_authority_class") or "CANARY")
        lifecycle_state = normalize_authority_lifecycle_state(policy.get("authority_lifecycle_state")) or "CERTIFIED"
        class_budget = AUTHORITY_CLASS_BUDGETS[authority_class]
        allowed_budget = max(0, min(to_int(policy.get("current_allowed_user_budget"), class_budget), class_budget))
        selected_before = len(selected)
        selected_after = selected[:allowed_budget] if selected_before > allowed_budget else list(selected)
        cap_applied = selected_after != selected
        prepared_exceeds_certified = AUTHORITY_CLASS_RANK[prepared_class] > AUTHORITY_CLASS_RANK[certified_class]
        if lifecycle_state in {"FROZEN", "REVOKED"} or allowed_budget <= 0:
            decision = "block_all_selected_moves_authority_budget_zero"
            action = "drop_selected_moves_before_restore_barrier_snapshot_apply"
            next_state = lifecycle_state if lifecycle_state in {"FROZEN", "REVOKED"} else "FROZEN"
        elif prepared_exceeds_certified:
            decision = "cap_prepared_authority_to_certified_evidence"
            action = "allow_only_certified_authority_budget"
            next_state = "CERTIFIED"
        elif cap_applied:
            decision = "cap_selected_moves_to_authority_budget"
            action = "drop_moves_above_current_authority_budget"
            next_state = authority_class
        else:
            decision = "selected_moves_within_authority_budget"
            action = "continue_existing_runtime_lifecycle"
            next_state = authority_class
        gate = {
            "schema_version": "v7.authority-budget-gate.v1",
            "enabled": bool_value(policy.get("enabled", True)),
            "truth_source": policy.get("truth_source", "existing_policy_files"),
            "runtime_calculation_mode": policy.get("runtime_calculation_mode", "prepared_and_certified_min_authority"),
            "authority_class": authority_class,
            "prepared_authority_class": prepared_class,
            "certified_authority_class": certified_class,
            "authority_lifecycle_state": lifecycle_state,
            "current_allowed_user_budget": allowed_budget,
            "next_authority_class": policy.get("next_authority_class") or AUTHORITY_CLASS_NEXT.get(authority_class, "CANARY"),
            "next_allowed_user_budget": to_int(policy.get("next_allowed_user_budget"), allowed_budget),
            "selected_moves_before_gate": selected_before,
            "selected_moves_after_gate": len(selected_after),
            "authority_cap_applied": cap_applied,
            "decision": decision,
            "action": action,
            "executor": "tools/v7-users-autoswitch",
            "trigger": "after_policy_selection_before_restore_barrier_snapshot_apply",
            "written_evidence": "plan.safety.authority_budget_gate",
            "blocked_actions": ["selected_moves_above_authority_budget", "apply_above_authority_budget"] if cap_applied or allowed_budget <= 0 else [],
            "next_state": next_state,
            "authority_lifecycle": {
                "schema_version": "v7.authority-governance-lifecycle.v1",
                "state": lifecycle_state,
                "prepared_authority_class": prepared_class,
                "certified_authority_class": certified_class,
                "runtime_authority_class": authority_class,
                "prepared_budget_ceiling": AUTHORITY_CLASS_BUDGETS[prepared_class],
                "certified_budget_ceiling": AUTHORITY_CLASS_BUDGETS[certified_class],
                "runtime_allowed_user_budget": allowed_budget,
                "prepared_exceeds_certified": prepared_exceeds_certified,
                "model": self._authority_lifecycle_model(),
                "certification_rules": self._authority_certification_rules(),
                "governance": self._authority_governance_policy(prepared_class, certified_class, lifecycle_state),
                "action_matrix": self._authority_full_action_matrix(),
                "new_truth_sources_created": False,
                "duplicate_systems_created": False,
            },
            "anti_flapping_policy": {
                "minimum_residence_seconds": to_int(policy.get("minimum_residence_seconds"), 300),
                "cooldown_seconds": to_int(policy.get("cooldown_seconds"), to_int(self.switch_policy.get("cooldown_seconds"), 180)),
                "sticky_protection": "existing_autoswitch_safety_state",
                "rollback_penalty_seconds": to_int(policy.get("rollback_penalty_seconds"), 3600),
                "promotion_delay_seconds": to_int(policy.get("promotion_delay_seconds"), 900),
                "demotion_delay_seconds": to_int(policy.get("demotion_delay_seconds"), 0),
            },
            "policy": {
                "evidence_floor": str(policy.get("evidence_floor") or ""),
                "class_budget_ceiling": class_budget,
                "new_truth_sources_created": False,
                "new_execution_path_created": False,
                "network_calls_in_apply_path": False,
                "history_scan_in_apply_path": False,
            },
            "action_matrix": self._authority_action_matrix(authority_class, allowed_budget),
        }
        if prepared_exceeds_certified:
            gate["blocked_actions"] = sorted(set(gate["blocked_actions"] + ["promotion_without_certification", "apply_above_certified_budget"]))
        if not gate["enabled"]:
            gate["decision"] = "authority_budget_gate_disabled_by_policy"
            gate["blocked_actions"] = ["disable_authority_budget_gate_in_production"]
            gate["selected_moves_after_gate"] = 0
            return [], gate
        return selected_after, gate

    def _recent_audit_records(self, limit: int = 200) -> list[dict[str, Any]]:
        path = self.event_dir / "switch-history.jsonl"
        try:
            lines = path.read_text(encoding="utf-8", errors="replace").splitlines()[-max(0, limit):]
        except OSError:
            return []
